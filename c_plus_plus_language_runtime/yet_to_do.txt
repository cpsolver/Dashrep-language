void parameterized_remove_extra_spaces(std::string& text_to_modify) {
    bool needs_normalization = false;

    if (!text_to_modify.empty() && std::isspace((unsigned char)text_to_modify.front())) {
        needs_normalization = true;
    } else if (!text_to_modify.empty() && std::isspace((unsigned char)text_to_modify.back())) {
        needs_normalization = true;
    } else {
        for (size_t i = 0; i < text_to_modify.size(); ++i) {
            if (std::isspace((unsigned char)text_to_modify[i]) &&
                (text_to_modify[i] != ' ' || (i + 1 < text_to_modify.size() &&
                std::isspace((unsigned char)text_to_modify[i + 1])))) {
                needs_normalization = true;
                break;
            }
        }
    }

    if (!needs_normalization) return;

    std::istringstream text_to_modify_as_stream(text_to_modify);
    std::string local_word;
    global_text_possibly_with_extra_spaces.clear();

    while (text_to_modify_as_stream >> local_word) {
        if (!global_text_possibly_with_extra_spaces.empty()) {
            global_text_possibly_with_extra_spaces += ' ';
        }
        global_text_possibly_with_extra_spaces += local_word;
    }

    text_to_modify = global_text_possibly_with_extra_spaces;
}
